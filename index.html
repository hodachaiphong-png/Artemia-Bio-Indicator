overlapping/stationary Artemia may be missed. class CentroidTracker: def __init__(self, max_distance=35, max_missing=8): self.next_id = 1 self.objects = {} self.missing = {} self.max_distance = max_distance self.max_missing = max_missing def update(self, detections): assigned = {} used = set() # Greedy nearest-neighbour association candidates = [] for oid, (ox, oy) in self.objects.items(): for j, (x, y, area, box) in enumerate(detections): d = float(np.hypot(ox-x, oy-y)) if d <= self.max_distance: candidates.append((d, oid, j)) for _, oid, j in sorted(candidates): if oid not in used and j not in assigned: assigned[j] = oid used.add(oid) for j, det in enumerate(detections): if j not in assigned: assigned[j] = self.next_id self.next_id += 1 new_objects, new_missing = {}, {} for j, (x, y, area, box) in enumerate(detections): oid = assigned[j] new_objects[oid] = (x, y) new_missing[oid] = 0 for oid in self.objects: if oid not in new_objects: m = self.missing.get(oid, 0) + 1 if m <= self.max_missing: new_objects[oid] = self.objects[oid] new_missing[oid] = m self.objects, self.missing = new_objects, new_missing return {assigned[j]: detections[j] for j in range(len(detections))} class ArtemiaAutoApp: def __init__(self, root): self.root = root root.title(APP_TITLE) root.geometry("820x620") root.minsize(720, 540) self.video_path = tk.StringVar() self.min_area = tk.IntVar(value=10) self.max_area = tk.IntVar(value=1800) self.sensitivity = tk.IntVar(value=18) self.frame_step = tk.IntVar(value=2) self.status = tk.StringVar(value="Chọn video để bắt đầu.") self.progress = tk.DoubleVar(value=0) self._build() def _build(self): main = ttk.Frame(self.root, padding=14) main.pack(fill="both", expand=True) ttk.Label(main, text="ARTEMIA BIO-INDICATOR V2 AUTO", font=("Segoe UI", 17, "bold")).pack(anchor="w") ttk.Label(main, text="Tự động phát hiện vùng chuyển động, gán ID và xuất video đánh dấu + CSV.", wraplength=760).pack(anchor="w", pady=(3, 12)) row = ttk.Frame(main);
